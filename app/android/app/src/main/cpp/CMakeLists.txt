cmake_minimum_required(VERSION 3.22)
project(signalsweep_detector LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(FW "${ROOT}/firmware/src")
set(JSON "${ROOT}/firmware/.pio/libdeps/tier1/ArduinoJson/src")
if(NOT EXISTS "${JSON}/ArduinoJson.h")
  include(FetchContent)
  FetchContent_Declare(arduinojson URL https://github.com/bblanchon/ArduinoJson/archive/refs/tags/v7.4.3.tar.gz)
  FetchContent_MakeAvailable(arduinojson)
  set(JSON "${arduinojson_SOURCE_DIR}/src")
endif()
add_library(signalsweep_detector SHARED detector.cpp "${FW}/opendroneid.c" "${FW}/odid_wifi.c" "${FW}/vendor/miniz_tinfl.c")
target_include_directories(signalsweep_detector PRIVATE "${FW}" "${JSON}")
target_compile_features(signalsweep_detector PRIVATE cxx_std_17)
target_compile_definitions(signalsweep_detector PRIVATE SIGNALSWEEP_PORTABLE=1)
target_link_libraries(signalsweep_detector m)
